// main.dart
import 'package:flutter/material.dart';
import 'app_bar.dart';
import 'bottom_navigation_bar.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: '老井无敌最俊朗',
      theme: ThemeData.dark().copyWith(
        textTheme: TextTheme(
          bodyLarge: TextStyle(fontSize: 14.0, color: Colors.white),
        ),
      ),
      home: MyHomePage(),
    );
  }
}

class MyHomePage extends StatefulWidget {
  @override
  _MyHomePageState createState() => _MyHomePageState();
}

class _MyHomePageState extends State<MyHomePage> {
  List<int> items = List.generate(50, (index) => index);
  ScrollController _scrollController = ScrollController();

  @override
  void initState() {
    super.initState();
    _scrollController.addListener(() {
      if (_scrollController.position.pixels ==
          _scrollController.position.maxScrollExtent) {
        setState(() {
          items.addAll(List.generate(10, (index) => items.length + index));
        });
      }
    });
  }

  Future<void> _refreshList() async {
    await Future.delayed(Duration(seconds: 2), () {
      setState(() {
        items = List.generate(50, (index) => index);
      });
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: buildAppBar(),
      body: Container(
        child: RefreshIndicator(
          onRefresh: _refreshList,
          child: ListView.builder(
            controller: _scrollController,
            itemCount: items.length,
            itemBuilder: (context, index) {
              return Container(
                height: 50.0,
                child: Text('${items[index]}'),
              );
            },
          ),
        ),
      ),
      bottomNavigationBar: buildBottomNavigationBar(),
    );
  }
}

这里使用了RefreshIndicator组件实现了下拉刷新功能，使用了ScrollController监听滑动到底部时加载更多数据。

  _MyHomePageState createState() => _MyHomePageState();
}

class _MyHomePageState extends State<MyHomePage> {
  ScrollController _scrollController = ScrollController();
  int _counter = 50;
  bool _isLoading = false;

  @override
  void dispose() {
    _scrollController.dispose();
    super.dispose();
  }

  void _incrementCounter() {
    setState(() {
      _counter++;
    });
  }

  void _loadData() async {
    _isLoading = true;
    await Future.delayed(Duration(seconds: 2), () {
      _scrollController.animateTo(
        _scrollController.position.pixels + 100,
        duration: Duration(microseconds: 200),
        curve: Curves.easeOut,
      );
      setState(() {
        _counter += 50;
        _isLoading = false;
      });
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text("老井无敌最俊朗"),
      ),
      body: RefreshIndicator(
        onRefresh: () async {
          await Future.delayed(Duration(seconds: 2), () {
            setState(() {
              _counter = 50;
            });
          });

          itemCount: _counter,
          itemBuilder: (context, index) {

              height: 50,
